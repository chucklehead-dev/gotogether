    field OnCommit func(ctx context.Context, partition, next int)

    method (*Reader) Poll(ctx context.Context) ([]simlog.Record, error)
type Writer struct
    method (*Writer) Close()

    method (*Writer) Produce(ctx context.Context, key string, value []byte, done func(error))
